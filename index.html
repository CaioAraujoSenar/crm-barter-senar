function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Produtores') || ss.getSheets()[0];
  const rows = sheet.getDataRange().getValues();
  if (rows.length < 2) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const headers = rows.shift();
  const data = rows.filter(r => r[0] !== '').map(r => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = r[i]);
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Produtores') || ss.getSheets()[0];
  const payload = JSON.parse(e.postData.contents);
  const headers = ['id','produtor','municipio','cpf_cnpj','fazenda','area_ha','passivo_prodes','empresa_regularizacao','endereco','telefone','termo_barter','status_crm','lat','lng','observacoes','updated_by','updated_at'];
  if (sheet.getLastRow() === 0) sheet.appendRow(headers);

  if (payload.action === 'sync_all' && Array.isArray(payload.records)) {
    sheet.clearContents();
    sheet.appendRow(headers);
    const matrix = payload.records.map(rec => headers.map(h => rec[h] ?? ''));
    if (matrix.length > 0) sheet.getRange(2, 1, matrix.length, headers.length).setValues(matrix);
  } else if (payload.action === 'delete' && payload.id !== undefined) {
    const rows = sheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(payload.id)) sheet.deleteRow(i + 1);
    }
  } else if (payload.record) {
    const rec = payload.record;
    const rows = sheet.getDataRange().getValues();
    const rowIndex = rows.findIndex((r, idx) => idx > 0 && String(r[0]) === String(rec.id));
    const rowValues = headers.map(h => rec[h] ?? '');
    if (rowIndex > 0) {
      sheet.getRange(rowIndex + 1, 1, 1, headers.length).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, timestamp: new Date().toISOString() }))
    .setMimeType(ContentService.MimeType.JSON);
}
